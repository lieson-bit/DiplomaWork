### **4. services/knapsack.service.ts** - Packing Algorithm
```typescript
export interface Item {
  id: string;
  weight: number; // in kg
  volume: number; // in m³
  value?: number; // optional for weighted scoring
  fragile: boolean;
  temperatureSensitive: boolean;
}

export interface VehicleCapacity {
  maxWeight: number; // kg
  maxVolume: number; // m³
  hasRefrigeration: boolean;
}

export class KnapsackService {
  // 0/1 Knapsack algorithm for optimal loading
  optimizeLoading(items: Item[], capacity: VehicleCapacity): {
    selectedItems: Item[];
    totalWeight: number;
    totalVolume: number;
    utilization: { weight: number; volume: number };
  } {
    // Filter out items that need special handling if vehicle doesn't support
    const feasibleItems = items.filter(item => {
      if (item.temperatureSensitive && !capacity.hasRefrigeration) {
        return false;
      }
      if (item.weight > capacity.maxWeight || item.volume > capacity.maxVolume) {
        return false; // Single item exceeds capacity
      }
      return true;
    });
    
    // Create weight-volume combined metric (pseudo-value)
    const n = feasibleItems.length;
    const dp: number[][] = Array(n + 1)
      .fill(0)
      .map(() => Array(capacity.maxWeight + 1).fill(0));
    
    // Track selected items
    const selected: boolean[][] = Array(n + 1)
      .fill(false)
      .map(() => Array(capacity.maxWeight + 1).fill(false));
    
    // Dynamic programming solution
    for (let i = 1; i <= n; i++) {
      const item = feasibleItems[i - 1];
      const itemWeight = Math.floor(item.weight);
      const itemValue = item.volume * 100; // Prioritize volume efficiency
      
      for (let w = 0; w <= capacity.maxWeight; w++) {
        if (itemWeight <= w) {
          const includeValue = dp[i - 1][w - itemWeight] + itemValue;
          if (includeValue > dp[i - 1][w]) {
            dp[i][w] = includeValue;
            selected[i][w] = true;
          } else {
            dp[i][w] = dp[i - 1][w];
          }
        } else {
          dp[i][w] = dp[i - 1][w];
        }
      }
    }
    
    // Backtrack to find selected items
    const selectedItems: Item[] = [];
    let w = capacity.maxWeight;
    for (let i = n; i > 0; i--) {
      if (selected[i][w]) {
        selectedItems.push(feasibleItems[i - 1]);
        w -= Math.floor(feasibleItems[i - 1].weight);
      }
    }
    
    const totalWeight = selectedItems.reduce((sum, item) => sum + item.weight, 0);
    const totalVolume = selectedItems.reduce((sum, item) => sum + item.volume, 0);
    
    return {
      selectedItems,
      totalWeight,
      totalVolume,
      utilization: {
        weight: (totalWeight / capacity.maxWeight) * 100,
        volume: (totalVolume / capacity.maxVolume) * 100
      }
    };
  }
  
  // First-Fit Decreasing algorithm for batch packing
  batchPackOrders(orders: any[], vehicles: any[]): Map<string, any[]> {
    // Sort orders by volume (largest first)
    const sortedOrders = [...orders].sort((a, b) => b.totalVolume - a.totalVolume);
    const assignments = new Map<string, any[]>();
    
    vehicles.forEach(vehicle => {
      assignments.set(vehicle.id, []);
      let remainingWeight = vehicle.maxWeight;
      let remainingVolume = vehicle.maxVolume;
      
      for (let i = 0; i < sortedOrders.length; i++) {
        const order = sortedOrders[i];
        if (!order.assigned && 
            order.totalWeight <= remainingWeight && 
            order.totalVolume <= remainingVolume) {
          
          assignments.get(vehicle.id)!.push(order);
          order.assigned = true;
          remainingWeight -= order.totalWeight;
          remainingVolume -= order.totalVolume;
        }
      }
    });
    
    return assignments;
  }
}
```

### **5. services/matching.service.ts** - Driver-Order Matching
```typescript
import { HttpClient } from '../utils/httpClient';
import { KnapsackService } from './knapsack.service';

export class MatchingService {
  private httpClient = new HttpClient();
  private knapsackService = new KnapsackService();
  
  async findBestDriver(order: any) {
    // 1. Get available drivers near pickup location
    const drivers = await this.httpClient.getAvailableDrivers(
      order.pickup_latitude,
      order.pickup_longitude
    );
    
    // 2. Filter drivers by capacity
    const suitableDrivers = [];
    
    for (const driver of drivers) {
      const vehicles = await this.httpClient.getDriverCapacity(driver.id);
      
      for (const vehicle of vehicles) {
        // Check if vehicle can handle order
        if (this.canVehicleHandleOrder(vehicle, order)) {
          suitableDrivers.push({
            driver,
            vehicle,
            distance: this.calculateDistance(
              vehicle.currentLocation,
              { lat: order.pickup_latitude, lng: order.pickup_longitude }
            ),
            rating: driver.rating
          });
        }
      }
    }
    
    // 3. Rank drivers by:
    //    - Proximity (40% weight)
    //    - Rating (30% weight)
    //    - Vehicle suitability (20% weight)
    //    - Availability (10% weight)
    
    const rankedDrivers = suitableDrivers
      .map(d => ({
        ...d,
        score: this.calculateDriverScore(d)
      }))
      .sort((a, b) => b.score - a.score);
    
    return rankedDrivers.length > 0 ? rankedDrivers[0] : null;
  }
  
  canVehicleHandleOrder(vehicle: any, order: any): boolean {
    return vehicle.maxWeight >= order.total_weight_kg &&
           vehicle.maxVolume >= order.total_volume_m3 &&
           vehicle.currentStatus === 'available' &&
           (!order.temperature_controlled || vehicle.hasRefrigeration);
  }
  
  calculateDistance(loc1: any, loc2: any): number {
    // Haversine formula implementation
    const R = 6371; // Earth's radius in km
    const dLat = this.toRad(loc2.lat - loc1.lat);
    const dLon = this.toRad(loc2.lng - loc1.lng);
    const a = 
      Math.sin(dLat/2) * Math.sin(dLat/2) +
      Math.cos(this.toRad(loc1.lat)) * Math.cos(this.toRad(loc2.lat)) *
      Math.sin(dLon/2) * Math.sin(dLon/2);
    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
    return R * c;
  }
  
  private toRad(degrees: number): number {
    return degrees * (Math.PI / 180);
  }
  
  private calculateDriverScore(driverInfo: any): number {
    // Normalize values
    const maxDistance = 50; // km
    const normalizedDistance = 1 - (Math.min(driverInfo.distance, maxDistance) / maxDistance);
    const normalizedRating = driverInfo.rating / 5;
    
    return (
      normalizedDistance * 0.4 +
      normalizedRating * 0.3 +
      driverInfo.vehicle.suitability * 0.2 +
      0.1 // Availability weight
    );
  }
}
```

### **6. services/pricing.service.ts** - Dynamic Pricing
```typescript
export class PricingService {
  calculateOrderPrice(order: any): {
    basePrice: number;
    distanceFee: number;
    weightFee: number;
    rushFee: number;
    fuelSurcharge: number;
    taxAmount: number;
    totalPrice: number;
  } {
    const basePrice = 5.0; // Minimum charge
    
    // Distance fee: $2 per km
    const distanceFee = (order.estimated_distance_km || 1) * 2;
    
    // Weight fee: $0.5 per kg after first 5kg
    const weightFee = Math.max(0, (order.total_weight_kg || 0) - 5) * 0.5;
    
    // Volume fee: $10 per m³
    const volumeFee = (order.total_volume_m3 || 0) * 10;
    
    // Rush fee: 20% extra for urgent orders
    const rushFee = order.priority === 'urgent' ? 
      (basePrice + distanceFee + weightFee + volumeFee) * 0.2 : 0;
    
    // Fuel surcharge: 10% of distance fee
    const fuelSurcharge = distanceFee * 0.1;
    
    const subtotal = basePrice + distanceFee + weightFee + volumeFee + rushFee + fuelSurcharge;
    
    // Tax: 10% VAT
    const taxAmount = subtotal * 0.1;
    
    return {
      basePrice,
      distanceFee,
      weightFee: weightFee + volumeFee,
      rushFee,
      fuelSurcharge,
      taxAmount,
      totalPrice: subtotal + taxAmount
    };
  }
  
  calculateBulkDiscount(ordersCount: number, totalPrice: number): {
    discountPercent: number;
    discountAmount: number;
    finalPrice: number;
  } {
    let discountPercent = 0;
    
    if (ordersCount >= 50) discountPercent = 15;
    else if (ordersCount >= 20) discountPercent = 10;
    else if (ordersCount >= 10) discountPercent = 5;
    
    const discountAmount = totalPrice * (discountPercent / 100);
    
    return {
      discountPercent,
      discountAmount,
      finalPrice: totalPrice - discountAmount
    };
  }
}
```

### **7. services/order.service.ts** - Main Business Logic
```typescript
import { KnapsackService } from './knapsack.service';
import { MatchingService } from './matching.service';
import { PricingService } from './pricing.service';
import { HttpClient } from '../utils/httpClient';
import { OrderRepository } from '../repositories/order.repository';

export class OrderService {
  private knapsackService = new KnapsackService();
  private matchingService = new MatchingService();
  private pricingService = new PricingService();
  private httpClient = new HttpClient();
  private orderRepository = new OrderRepository();
  
  async createOrder(customerId: string, orderData: any) {
    // 1. Validate customer exists
    await this.httpClient.validateUser(customerId);
    
    // 2. Calculate price
    const price = this.pricingService.calculateOrderPrice(orderData);
    
    // 3. Create order in database
    const order = await this.orderRepository.create({
      ...orderData,
      customer_id: customerId,
      ...price,
      status: 'pending'
    });
    
    // 4. Queue for driver matching
    await this.queueForMatching(order.id);
    
    return order;
  }
  
  async createBulkOrder(customerId: string, ordersData: any[]) {
    // Process multiple orders with batch optimizations
    const processedOrders = [];
    
    for (const orderData of ordersData) {
      try {
        const order = await this.createOrder(customerId, orderData);
        processedOrders.push(order);
      } catch (error) {
        // Log error but continue processing other orders
        console.error(`Failed to create order: ${error.message}`);
      }
    }
    
    return {
      success: processedOrders.length,
      failed: ordersData.length - processedOrders.length,
      orders: processedOrders
    };
  }
  
  async processExcelUpload(fileBuffer: Buffer, customerId: string) {
    // Parse Excel file and create bulk orders
    const ordersData = this.parseExcel(fileBuffer);
    return this.createBulkOrder(customerId, ordersData);
  }
  
  async assignDriver(orderId: string) {
    // 1. Get order details
    const order = await this.orderRepository.findById(orderId);
    
    // 2. Find best driver using matching service
    const driverMatch = await this.matchingService.findBestDriver(order);
    
    if (!driverMatch) {
      throw new Error('No suitable driver found');
    }
    
    // 3. Assign driver and update order
    await this.orderRepository.update(orderId, {
      driver_id: driverMatch.driver.id,
      status: 'assigned',
      assigned_at: new Date()
    });
    
    // 4. Notify driver (call driver-service)
    await this.httpClient.notifyDriverAssignment(
      driverMatch.driver.id,
      orderId
    );
    
    return {
      order,
      driver: driverMatch.driver,
      vehicle: driverMatch.vehicle
    };
  }
  
  async updateOrderStatus(orderId: string, status: string, notes?: string) {
    // Update status with history tracking
    await this.orderRepository.updateStatus(orderId, status, notes);
    
    // Real-time tracking event
    await this.trackEvent(orderId, `status_changed_to_${status}`);
    
    return { success: true, status };
  }
  
  async getOrderWithOptimalPacking(orderId: string) {
    const order = await this.orderRepository.findById(orderId);
    const orderItems = await this.orderRepository.getOrderItems(orderId);
    
    if (order.driver_id) {
      const vehicle = await this.httpClient.getDriverCapacity(order.driver_id);
      
      // Optimize loading using knapsack
      const optimization = this.knapsackService.optimizeLoading(
        orderItems.map(item => ({
          id: item.id,
          weight: item.weight_per_item_kg,
          volume: this.calculateVolume(item),
          fragile: item.fragile,
          temperatureSensitive: item.temperature_sensitive
        })),
        {
          maxWeight: vehicle.maxWeight,
          maxVolume: vehicle.maxVolume,
          hasRefrigeration: vehicle.hasRefrigeration
        }
      );
      
      return {
        order,
        packingOptimization: optimization,
        suggestedLoading: this.generateLoadingPlan(optimization.selectedItems)
      };
    }
    
    return { order, items: orderItems };
  }
  
  private calculateVolume(item: any): number {
    if (item.dimensions_length_cm && item.dimensions_width_cm && item.dimensions_height_cm) {
      return (item.dimensions_length_cm * item.dimensions_width_cm * item.dimensions_height_cm) / 1000000; // Convert to m³
    }
    return 0.01; // Default volume if dimensions not provided
  }
  
  private generateLoadingPlan(items: any[]): any[] {
    // Generate loading sequence (heaviest at bottom, fragile on top)
    return items
      .sort((a, b) => {
        if (a.fragile && !b.fragile) return 1;
        if (!a.fragile && b.fragile) return -1;
        return b.weight - a.weight; // Heaviest first
      })
      .map((item, index) => ({
        item,
        position: index + 1,
        layer: Math.floor(index / 3) + 1, // 3 items per layer
        notes: item.fragile ? 'Handle with care' : 'Standard handling'
      }));
  }
  
  private async queueForMatching(orderId: string) {
    await this.orderRepository.addToAssignmentQueue(orderId);
  }
  
  private async trackEvent(orderId: string, eventType: string) {
    await this.orderRepository.createTrackingEvent(orderId, eventType);
  }
  
  private parseExcel(buffer: Buffer): any[] {

  }
}
```

```typescript
import { Request, Response, NextFunction } from 'express';
import { KnapsackService } from '../services/knapsack.service';

export const validateOrderCapacity = async (
  req: Request,
  res: Response,
  next: NextFunction
) => {
  const orderData = req.body;
  const knapsackService = new KnapsackService();
  
  // Validate individual item capacities
  if (orderData.order_items) {
    for (const item of orderData.order_items) {
      if (item.weight_per_item_kg > 50) { // Max 50kg per item
        return res.status(400).json({
          error: `Item "${item.item_name}" exceeds maximum weight of 50kg`
        });
      }
      
      // Calculate volume
      const volume = (item.dimensions_length_cm * 
                     item.dimensions_width_cm * 
                     item.dimensions_height_cm) / 1000000;
      
      if (volume > 1) { // Max 1m³ per item
        return res.status(400).json({
          error: `Item "${item.item_name}" exceeds maximum volume of 1m³`
        });
      }
    }
  }
  
  next();
};
```

```typescript
import { HttpClient } from '../utils/httpClient';
import { OrderRepository } from '../repositories/order.repository';

export class PaymentService {
  private httpClient = new HttpClient();
  private orderRepository = new OrderRepository();
  
  async processOrderCompletion(orderId: string): Promise<{
    success: boolean;
    paymentDetails: any;
    balanceUpdates: any;
  }> {
    try {
      // 1. Get order details
      const order = await this.orderRepository.findById(orderId);
      
      // 2. Verify order is delivered
      if (order.status !== 'delivered') {
        throw new Error('Order must be delivered before payment processing');
      }
      
      // 3. Calculate driver earnings (total - platform fee)
      const driverEarnings = order.total_price - order.platform_fee;
      
      // 4. Update customer balance (deduct total)
      const customerUpdate = await this.httpClient.updateCustomerBalance(
        order.customer_id,
        -order.total_price, // Negative = debit from customer
        'order_payment',
        orderId
      );
      
      // 5. Update driver balance (add earnings)
      const driverUpdate = await this.httpClient.updateDriverBalance(
        order.driver_id,
        driverEarnings, // Positive = credit to driver
        'order_completion',
        orderId
      );
      
      // 6. Record payment transaction
      const paymentRecord = await this.orderRepository.createPaymentRecord({
        order_id: orderId,
        amount: order.total_price,
        fee_amount: order.platform_fee,
        net_amount: driverEarnings,
        payment_method: 'wallet',
        status: 'completed',
        customer_balance_before: customerUpdate.previousBalance,
        customer_balance_after: customerUpdate.newBalance,
        driver_balance_before: driverUpdate.previousBalance,
        driver_balance_after: driverUpdate.newBalance
      });
      
      // 7. Update order with payment status
      await this.orderRepository.update(orderId, {
        payment_status: 'completed',
        payment_processed_at: new Date(),
        driver_earnings: driverEarnings,
        customer_balance_updated: true,
        driver_balance_updated: true,
        status: 'completed',
        completed_at: new Date()
      });
      
      // 8. Send notifications
      await this.sendPaymentNotifications(order, driverEarnings);
      
      return {
        success: true,
        paymentDetails: paymentRecord,
        balanceUpdates: {
          customer: customerUpdate,
          driver: driverUpdate
        }
      };
      
    } catch (error) {
      // Retry logic for balance updates
      await this.handleFailedPayment(orderId, error);
      throw error;
    }
  }
  
  async verifyCustomerBalance(customerId: string, amount: number): Promise<boolean> {
    try {
      const customer = await this.httpClient.getCustomerBalance(customerId);
      return customer.balance >= amount;
    } catch (error) {
      throw new Error(`Failed to verify customer balance: ${error.message}`);
    }
  }
  
  async refundOrder(orderId: string, reason: string): Promise<any> {
    const order = await this.orderRepository.findById(orderId);
    
    if (order.payment_status !== 'completed') {
      throw new Error('Cannot refund order that hasn\'t been paid');
    }
    
    // Refund customer
    const customerRefund = await this.httpClient.updateCustomerBalance(
      order.customer_id,
      order.total_price, // Positive = refund to customer
      'order_refund',
      orderId
    );
    
    // Deduct from driver (if already paid)
    if (order.driver_balance_updated) {
      await this.httpClient.updateDriverBalance(
        order.driver_id,
        -order.driver_earnings, // Negative = deduct from driver
        'order_refund',
        orderId
      );
    }
    
    // Update order
    await this.orderRepository.update(orderId, {
      payment_status: 'refunded',
      status: 'cancelled',
      cancelled_at: new Date()
    });
    
    return { success: true, refundedAmount: order.total_price };
  }
  
  private async sendPaymentNotifications(order: any, driverEarnings: number) {
    // Notify customer
    await this.httpClient.sendNotification(order.customer_id, {
      type: 'payment_completed',
      title: 'Payment Processed',
      message: `Payment of $${order.total_price} for order #${order.order_number} has been processed.`,
      data: { orderId: order.id }
    });
    
    // Notify driver
    await this.httpClient.sendNotification(order.driver_id, {
      type: 'earnings_received',
      title: 'Earnings Credited',
      message: `You've earned $${driverEarnings} for completing order #${order.order_number}.`,
      data: { orderId: order.id, earnings: driverEarnings }
    });
  }
  
  private async handleFailedPayment(orderId: string, error: any) {
    // Increment retry count
    const attempts = await this.orderRepository.incrementBalanceUpdateAttempts(orderId);
    
    if (attempts >= 3) {
      // Mark for manual review after 3 failed attempts
      await this.orderRepository.update(orderId, {
        payment_status: 'failed',
        internal_notes: `Payment failed after ${attempts} attempts: ${error.message}`
      });
    }
  }
}
```

### **3. services/routing.service.ts** - Shortest Path Calculation
```typescript
export class RoutingService {
  async calculateOptimalRoute(
    pickupLat: number,
    pickupLng: number,
    deliveryLat: number,
    deliveryLng: number
  ): Promise<{
    distance: number; // in km
    duration: number; // in minutes
    polyline: string; // encoded route
    steps: Array<{lat: number, lng: number}>;
  }> {
    // Use OSRM or Google Maps API for real routing
    // For MVP, use Haversine + traffic factor
    
    // 1. Calculate straight-line distance
    const directDistance = this.calculateHaversineDistance(
      pickupLat, pickupLng,
      deliveryLat, deliveryLng
    );
    
    // 2. Add road factor (typically 1.3x for road distance)
    const roadDistance = directDistance * 1.3;
    
    // 3. Calculate estimated duration (assuming 30km/h average in city)
    const durationMinutes = (roadDistance / 30) * 60;
    
    // 4. Apply traffic factor based on time of day
    const trafficFactor = this.getTrafficFactor();
    const finalDuration = durationMinutes * trafficFactor;
    
    // 5. Generate simplified polyline
    const polyline = this.encodePolyline([
      { lat: pickupLat, lng: pickupLng },
      { lat: deliveryLat, lng: deliveryLng }
    ]);
    
    return {
      distance: roadDistance,
      duration: finalDuration,
      polyline,
      steps: [
        { lat: pickupLat, lng: pickupLng },
        { lat: deliveryLat, lng: deliveryLng }
      ]
    };
  }
  
  async getLiveRoute(driverLat: number, driverLng: number, order: any) {
    // Calculate ETA based on current position
    const remainingDistance = this.calculateHaversineDistance(
      driverLat, driverLng,
      order.delivery_latitude, order.delivery_longitude
    );
    
    // Real-time ETA calculation
    const currentSpeed = await this.getCurrentSpeed(driverLat, driverLng);
    const etaMinutes = (remainingDistance / currentSpeed) * 60;
    
    return {
      remainingDistance,
      etaMinutes,
      currentLocation: { lat: driverLat, lng: driverLng },
      isOnRoute: this.isOnRoute(driverLat, driverLng, order.route_polyline)
    };
  }
  
  private calculateHaversineDistance(lat1: number, lon1: number, lat2: number, lon2: number): number {
    const R = 6371; // Earth's radius in km
    const dLat = this.toRad(lat2 - lat1);
    const dLon = this.toRad(lon2 - lon1);
    const a = 
      Math.sin(dLat/2) * Math.sin(dLat/2) +
      Math.cos(this.toRad(lat1)) * Math.cos(this.toRad(lat2)) *
      Math.sin(dLon/2) * Math.sin(dLon/2);
    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
    return R * c;
  }
  
  private getTrafficFactor(): number {
    const hour = new Date().getHours();
    // Peak hours: 7-9am, 4-7pm
    if ((hour >= 7 && hour <= 9) || (hour >= 16 && hour <= 19)) {
      return 1.5; // 50% longer due to traffic
    }
    return 1.1; // Normal traffic
  }
  
  private toRad(degrees: number): number {
    return degrees * (Math.PI / 180);
  }
  
  private encodePolyline(points: Array<{lat: number, lng: number}>): string {
    // Simple polyline encoding for MVP
    return points.map(p => `${p.lat},${p.lng}`).join('|');
  }
  
  private async getCurrentSpeed(lat: number, lng: number): number {
    // Mock: average city speed based on time
    const hour = new Date().getHours();
    if (hour >= 22 || hour <= 5) return 40; // Night: faster
    if (hour >= 7 && hour <= 9) return 15;  // Morning rush: slower
    if (hour >= 16 && hour <= 19) return 20; // Evening rush
    return 30; // Normal hours
  }
  
  private isOnRoute(lat: number, lng: number, polyline: string): boolean {
    // Simplified check: within 100m of any route point
    const points = polyline.split('|').map(p => {
      const [pointLat, pointLng] = p.split(',').map(Number);
      return { lat: pointLat, lng: pointLng };
    });
    
    for (const point of points) {
      const distance = this.calculateHaversineDistance(lat, lng, point.lat, point.lng);
      if (distance < 0.1) return true; // 100 meters
    }
    
    return false;
  }
}
```

```typescript
export class OrderService {
  private knapsackService = new KnapsackService();
  private matchingService = new MatchingService();
  private pricingService = new PricingService();
  private routingService = new RoutingService();
  private trackingService = new TrackingService();
  private paymentService = new PaymentService();
  private notificationService = new NotificationService();
  private httpClient = new HttpClient();
  private orderRepository = new OrderRepository();
  
  async createOrder(customerId: string, orderData: any) {
    // 1. Validate customer exists and has sufficient balance
    const customer = await this.httpClient.getCustomerDetails(customerId);
    
    // 2. Calculate route and pricing
    const route = await this.routingService.calculateOptimalRoute(
      orderData.pickup_latitude,
      orderData.pickup_longitude,
      orderData.delivery_latitude,
      orderData.delivery_longitude
    );
    
    const price = this.pricingService.calculateOrderPrice({
      ...orderData,
      distance_km: route.distance
    });
    
    // 3. Verify customer can afford the order
    const canAfford = await this.paymentService.verifyCustomerBalance(
      customerId,
      price.total_price
    );
    
    if (!canAfford) {
      throw new Error('Insufficient balance. Please top up your wallet.');
    }
    
    // 4. Create order with all calculated data
    const order = await this.orderRepository.create({
      ...orderData,
      customer_id: customerId,
      estimated_distance_km: route.distance,
      estimated_duration_minutes: route.duration,
      route_polyline: route.polyline,
      ...price,
      status: 'pending'
    });
    
    // 5. Immediately queue for driver matching with knapsack optimization
    await this.queueForMatching(order.id);
    
    // 6. Notify customer
    await this.notificationService.notifyOrderCreated(customerId, order);
    
    return order;
  }
  
  async matchOrderWithDriver(orderId: string) {
    const order = await this.orderRepository.findById(orderId);
    
    // 1. Get available drivers near pickup
    const drivers = await this.httpClient.getAvailableDrivers(
      order.pickup_latitude,
      order.pickup_longitude
    );
    
    // 2. Filter by vehicle capacity using knapsack algorithm
    const suitableDrivers = [];
    
    for (const driver of drivers) {
      const vehicles = await this.httpClient.getDriverCapacity(driver.id);
      
      for (const vehicle of vehicles) {
        // Check if vehicle can optimally pack order items
        const items = await this.orderRepository.getOrderItems(orderId);
        const packingResult = this.knapsackService.optimizeLoading(
          items.map(item => ({
            id: item.id,
            weight: item.weight_per_item_kg,
            volume: this.calculateVolume(item),
            fragile: item.fragile,
            temperatureSensitive: item.temperature_sensitive
          })),
          {
            maxWeight: vehicle.maxWeight,
            maxVolume: vehicle.maxVolume,
            hasRefrigeration: vehicle.hasRefrigeration
          }
        );
        
        if (packingResult.selectedItems.length === items.length) {
          // All items can be optimally packed
          suitableDrivers.push({
            driver,
            vehicle,
            packingEfficiency: packingResult.utilization,
            distance: this.calculateDistance(vehicle.currentLocation, {
              lat: order.pickup_latitude,
              lng: order.pickup_longitude
            })
          });
        }
      }
    }
    
    // 3. Select best driver (proximity + packing efficiency)
    if (suitableDrivers.length > 0) {
      const bestDriver = suitableDrivers.sort((a, b) => {
        const scoreA = (1 / a.distance) * a.packingEfficiency.volume;
        const scoreB = (1 / b.distance) * b.packingEfficiency.volume;
        return scoreB - scoreA;
      })[0];
      
      // 4. Assign driver and notify
      await this.orderRepository.update(orderId, {
        driver_id: bestDriver.driver.id,
        status: 'matched',
        matched_at: new Date()
      });
      
      // 5. Send push notification to driver
      await this.notificationService.notifyDriverAssignment(
        bestDriver.driver.id,
        order
      );
      
      return {
        order,
        driver: bestDriver.driver,
        vehicle: bestDriver.vehicle,
        packingPlan: this.generateLoadingPlan(bestDriver.packingEfficiency)
      };
    }
    
    throw new Error('No suitable driver found with optimal packing capacity');
  }
  
  async driverAcceptsOrder(orderId: string, driverId: string) {
    const order = await this.orderRepository.findById(orderId);
    
    if (order.driver_id !== driverId) {
      throw new Error('Driver not assigned to this order');
    }
    
    // Update order status
    await this.orderRepository.update(orderId, {
      status: 'driver_accepted',
      accepted_at: new Date()
    });
    
    // Notify customer
    await this.notificationService.notifyCustomerDriverAccepted(
      order.customer_id,
      order
    );
    
    return { success: true };
  }
  
  async startPickup(orderId: string, driverId: string, location: any) {
    const order = await this.orderRepository.findById(orderId);
    
    await this.orderRepository.update(orderId, {
      status: 'pickup_started',
      pickup_started_at: new Date(),
      driver_current_lat: location.latitude,
      driver_current_lng: location.longitude,
      driver_last_updated: new Date()
    });
    
    // Start real-time tracking
    await this.trackingService.startTracking(orderId, driverId);
    
    return { success: true };
  }
  
  async updateDriverLocation(orderId: string, location: any) {
    const order = await this.orderRepository.findById(orderId);
    
    // 1. Update current location
    await this.orderRepository.update(orderId, {
      driver_current_lat: location.latitude,
      driver_current_lng: location.longitude,
      driver_last_updated: new Date()
    });
    
    // 2. Log location for tracking

      order_id: orderId,
      driver_id: order.driver_id,
      latitude: location.latitude,
      longitude: location.longitude,
      speed: location.speed,
      bearing: location.bearing
    });
    
    // 3. Calculate ETA and broadcast to customer
    const routeUpdate = await this.routingService.getLiveRoute(
      location.latitude,
      location.longitude,
      order
    );
    
    // 4. Broadcast via WebSocket
    await this.trackingService.broadcastLocationUpdate({
      orderId,
      location: { lat: location.latitude, lng: location.longitude },
      eta: routeUpdate.etaMinutes,
      remainingDistance: routeUpdate.remainingDistance
    });
    
    return routeUpdate;
  }
  
  async markOrderInTransit(orderId: string) {
    await this.orderRepository.update(orderId, {
      status: 'in_transit',
      in_transit_at: new Date()
    });
    
    return { success: true };
  }
  
  async markOrderArrived(orderId: string) {
    await this.orderRepository.update(orderId, {
      status: 'arrived',
      arrived_at: new Date()
    });
    
    // Notify customer driver has arrived
    const order = await this.orderRepository.findById(orderId);
    await this.notificationService.notifyCustomerDriverArrived(
      order.customer_id,
      order
    );
    
    return { success: true };
  }
  
  async markOrderDelivered(orderId: string, proof?: any) {
    const order = await this.orderRepository.findById(orderId);
    
    // 1. Update order status
    await this.orderRepository.update(orderId, {
      status: 'delivered',
      delivered_at: new Date(),
      actual_distance_km: await this.calculateActualDistance(orderId),
      actual_duration_minutes: this.calculateActualDuration(order)
    });
    
    // 2. Process payment
    const paymentResult = await this.paymentService.processOrderCompletion(orderId);
    
    // 3. Send delivery confirmation
    await this.notificationService.notifyOrderDelivered(
      order.customer_id,
      order.driver_id,
      order
    );
    
    return {
      success: true,
      payment: paymentResult
    };
  }
  
  private async calculateActualDistance(orderId: string): Promise<number> {
    const trackingPoints = await this.trackingService.getTrackingPoints(orderId);
    
    if (trackingPoints.length < 2) return 0;
    
    let totalDistance = 0;
    for (let i = 1; i < trackingPoints.length; i++) {
      totalDistance += this.calculateHaversineDistance(
        trackingPoints[i-1].latitude,
        trackingPoints[i-1].longitude,
        trackingPoints[i].latitude,
        trackingPoints[i].longitude
      );
    }
    
    return totalDistance;
  }
  
  private calculateActualDuration(order: any): number {
    if (!order.pickup_started_at || !order.delivered_at) return 0;
    
    const start = new Date(order.pickup_started_at).getTime();
    const end = new Date(order.delivered_at).getTime();
    
    return Math.round((end - start) / (1000 * 60)); // minutes
  }
  
  private generateLoadingPlan(packingResult: any) {
    return {
      loadingSequence: packingResult.selectedItems
        .sort((a, b) => b.weight - a.weight) // Heaviest first
        .map((item, index) => ({
          itemId: item.id,
          position: index + 1,
          layer: Math.floor(index / 3) + 1,
          notes: item.fragile ? 'TOP LAYER - HANDLE WITH CARE' : 'Standard'
        })),
      weightUtilization: `${packingResult.utilization.weight.toFixed(1)}%`,
      volumeUtilization: `${packingResult.utilization.volume.toFixed(1)}%`,
      efficiencyScore: (packingResult.utilization.weight + packingResult.utilization.volume) / 2
    };
  }
}
```

### **7. services/tracking.service.ts** - Real-time Location Broadcasting
```typescript
import WebSocket from 'ws';
import { Server } from 'http';

export class TrackingService {
  private wss: WebSocket.Server;
  private connections: Map<string, WebSocket> = new Map();
  
  constructor(server: Server) {
    this.wss = new WebSocket.Server({ server });
    this.setupWebSocket();
  }
  
  private setupWebSocket() {
    this.wss.on('connection', (ws, req) => {
      const orderId = req.url?.split('=')[1];
      
      if (orderId) {
        this.connections.set(orderId, ws);
        
        ws.on('close', () => {
          this.connections.delete(orderId);
        });
      }
    });
  }
  
  async broadcastLocationUpdate(data: {
    orderId: string;
    location: { lat: number; lng: number };
    eta: number;
    remainingDistance: number;
  }) {
    const ws = this.connections.get(data.orderId);
    
    if (ws && ws.readyState === WebSocket.OPEN) {
      ws.send(JSON.stringify({
        type: 'location_update',
        data: {
          timestamp: new Date().toISOString(),
          location: data.location,
          eta: data.eta,
          remainingDistance: data.remainingDistance
        }
      }));
    }
    
    // Also store in database for history
    await this.logBroadcast(data);
  }
  
  async startTracking(orderId: string, driverId: string) {
    // Initialize tracking session
    await this.logLocation({
      order_id: orderId,
      driver_id: driverId,
      event_type: 'tracking_started',
      timestamp: new Date()
    });
  }
  
  async logLocation(data: any) {
    // Store location in database
    await this.trackingRepository.create(data);
  }
  
  async getLiveTracking(orderId: string) {
    const locations = await this.trackingRepository.getRecentLocations(orderId, 50);
    
    return {
      orderId,
      locations,
      lastUpdated: locations.length > 0 ? locations[0].timestamp : null
    };
  }
}
```

## **Complete Order Flow Summary**

### **Order Lifecycle with Payment Integration:**
```
1. CREATE ORDER
   → Customer provides pickup/delivery addresses + package details
   → System calculates shortest route + pricing
   → Verifies customer balance
   → Creates order in 'pending' status

2. KNAPSACK MATCHING
   → System finds available drivers near pickup
   → Applies knapsack algorithm to find optimal vehicle packing
   → Selects driver with best packing efficiency + proximity
   → Sends push notification to driver

3. DRIVER ACCEPTANCE
   → Driver reviews order details and accepts
   → Status changes to 'driver_accepted'
   → Customer notified

4. PICKUP & TRACKING
   → Driver goes to pickup location
   → Real-time GPS tracking starts
   → Customer sees driver on map
   → Status changes to 'pickup_started'

5. IN TRANSIT
   → Driver loads packages (knapsack optimized loading plan)
   → Starts delivery with shortest route
   → Real-time location updates every 30 seconds
   → Customer sees ETA and live tracking

6. DELIVERY & PAYMENT
   → Driver arrives at destination
   → Customer confirms delivery
   → Status changes to 'delivered'
   → Automatic payment processing:
     • Customer balance: -total_price
     • Driver balance: +earnings (total_price - platform_fee)
   → Status changes to 'completed'

7. COMPLETION
   → Both parties receive payment notifications
   → Driver rating system triggered
   → Order added to history
   → Analytics updated
```
